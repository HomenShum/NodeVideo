// voice.mjs — narrate the storyboard with a LOCAL TTS, aligned to the captions.
//
// THE SCRIPT IS THE CAPTIONS. There is no second script file, deliberately: the
// fastest way to ship a demo whose voice contradicts its subtitles is to keep the
// two in separate documents and edit one of them. Change a caption in the spec
// and the narration changes with it, or neither changes.
//
// WHY PIPER, MEASURED RATHER THAN CHOSEN FROM A SHORTLIST.
//
// The brief was a free, small, local replacement for ElevenLabs. Kokoro-82M is
// the usual first recommendation and it SEGFAULTED here on synthesis after
// installing cleanly -- which is the difference between a search result and a
// measurement. Piper runs: a 63MB onnx voice, no API key, no network at
// synthesis time, and it produced 4.27s of speech from a 13-word line on CPU.
//
//   python -m piper.download_voices en_US-lessac-medium     (once, ~63MB)
//   node voice.mjs --id TShero --out out/trialscope.vo.wav
//
import { execFileSync } from 'node:child_process';
import { existsSync, mkdirSync, readFileSync, writeFileSync } from 'node:fs';
import { COLLAB_WALKTHROUGHS } from './src/walkthrough.collab.data.js';

const argv = process.argv.slice(2);
const flag = (n, d) => {
  const i = argv.indexOf(`--${n}`);
  return i >= 0 && argv[i + 1] ? argv[i + 1] : d;
};

const id = flag('id');
const out = flag('out');
const voice = flag('voice', 'en_US-lessac-medium');
const fps = Number(flag('fps', '30'));
const MAX_TEMPO = Number(flag('max-tempo', '1.3')); // beyond this it sounds rushed
if (!id || !out) {
  console.error(
    'usage: node voice.mjs --id <SpecId> --out <file.wav> [--voice en_US-lessac-medium]',
  );
  process.exit(1);
}

const wt = COLLAB_WALKTHROUGHS.find((w) => w.id === id);
if (!wt) {
  console.error(`no walkthrough "${id}"`);
  process.exit(1);
}

const tmp = 'out/.vo';
mkdirSync(tmp, { recursive: true });

// Captions are written to be READ, not spoken: "--" is a visual dash, and a TTS
// says it as "minus minus" or swallows the pause entirely.
const speakable = (s) =>
  s
    .replace(/\s--\s/g, ', ')
    .replace(/\s+/g, ' ')
    .trim();

let t = 0;
const lines = [];
wt.steps.forEach((s, i) => {
  const at = t / fps;
  const dur = (s.hold || 60) / fps;
  t += s.hold || 60;
  if (s.caption) lines.push({ i, at, dur, text: speakable(s.caption) });
});
const DUR = t / fps;

const sh = (cmd, args) =>
  execFileSync(cmd, args, { encoding: 'utf8', stdio: ['pipe', 'pipe', 'pipe'] });
const dur = (f) =>
  Number(
    sh('ffprobe', ['-v', 'error', '-show_entries', 'format=duration', '-of', 'csv=p=0', f]).trim(),
  );

// --------------------------------------------------------------- synthesise
for (const l of lines) {
  const raw = `${tmp}/${id}-${l.i}.wav`;
  // Piper splits on sentence boundaries, and a caption made of very short
  // fragments ("Any condition. Any breakdown. Same receipts.") produces a
  // degenerate phoneme tensor and an onnxruntime Reshape failure. Joining the
  // fragments into one utterance fixes it, and reads better anyway -- three
  // full stops in four seconds is a caption rhythm, not a speaking one.
  const attempts = [l.text, l.text.replace(/\.\s+/g, ', ').replace(/,\s*$/, '.')];
  let ok = false;
  let lastErr = '';
  for (const text of attempts) {
    try {
      execFileSync('python', ['-m', 'piper', '--model', voice, '--output-file', raw], {
        input: text,
        stdio: ['pipe', 'ignore', 'pipe'],
        encoding: 'utf8',
      });
      l.spoken = text;
      ok = true;
      break;
    } catch (e) {
      lastErr = String(e.stderr || e.message);
    }
  }
  if (!ok) {
    console.error(`[voice] step ${l.i} FAILED after ${attempts.length} attempts: "${l.text}"`);
    console.error(lastErr.split(String.fromCharCode(0)).join('').slice(0, 300));
    process.exit(1);
  }
  l.raw = raw;
  l.rawDur = dur(raw);
  // A line longer than its step would talk over the next caption, which is the
  // one desync a viewer always notices. Compress time up to MAX_TEMPO; past that
  // the fix is a shorter caption, not a faster reader, so it is reported instead
  // of hidden.
  l.tempo = Math.min(MAX_TEMPO, Math.max(1, l.rawDur / (l.dur - 0.12)));
  l.over = l.rawDur / l.tempo > l.dur;
}

// PACING IS DOWNSTREAM OF NARRATION. A cut timed for reading captions is too
// fast to speak: 9 of 21 lines here overran their hold even at 1.3x. Rushing the
// reader to fit the picture is the wrong direction -- the picture is a still
// frame held for N frames, and holding it longer costs nothing but runtime. So
// the measured speech duration is written out as the hold each step NEEDS, and
// the storyboard is re-timed from it.
writeFileSync(
  out.replace(/\.wav$/, '.holds.json'),
  JSON.stringify(
    Object.fromEntries(lines.map((l) => [l.i, Math.ceil((l.rawDur / 1.12 + 0.45) * fps)])),
    null,
    1,
  ),
);

const overruns = lines.filter((l) => l.over);
if (overruns.length) {
  console.log(
    `[voice] ${overruns.length}/${lines.length} lines still overrun their step at ${MAX_TEMPO}x:`,
  );
  for (const l of overruns) {
    console.log(
      `  step ${l.i}: ${(l.rawDur / l.tempo).toFixed(1)}s of speech in a ${l.dur.toFixed(1)}s hold — "${l.text.slice(0, 52)}..."`,
    );
  }
}

// ------------------------------------------------- lay each line at its step
const inputs = lines.flatMap((l) => ['-i', l.raw]);
const chains = lines
  .map((l, k) => {
    const delay = Math.round(l.at * 1000);
    const tempo = l.tempo > 1.001 ? `atempo=${l.tempo.toFixed(3)},` : '';
    return `[${k}:a]${tempo}aresample=48000,adelay=${delay}|${delay},apad[v${k}]`;
  })
  .join(';');
const mixIn = lines.map((_, k) => `[v${k}]`).join('');

execFileSync(
  'ffmpeg',
  [
    '-y',
    '-loglevel',
    'error',
    ...inputs,
    '-filter_complex',
    // Speech wants a narrow band and a little presence; loudnorm at -16 leaves
    // headroom for the music to sit under it after ducking.
    `${chains};${mixIn}amix=inputs=${lines.length}:normalize=0:duration=longest[m];[m]highpass=f=90,lowpass=f=11000,acompressor=threshold=-18dB:ratio=3:attack=8:release=180,atrim=0:${(DUR + 1).toFixed(2)},loudnorm=I=-16:TP=-1.5:LRA=9[a]`,
    '-map',
    '[a]',
    '-ac',
    '2',
    '-ar',
    '48000',
    out,
  ],
  { stdio: 'inherit' },
);

console.log(`[voice] ${out} — ${lines.length} lines, ${DUR.toFixed(1)}s, voice=${voice}`);
