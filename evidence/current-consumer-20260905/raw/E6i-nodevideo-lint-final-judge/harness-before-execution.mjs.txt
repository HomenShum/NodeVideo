function normal(x){
  if(x===undefined)return {$undefined:true};
  if(typeof x==='number'&&!Number.isFinite(x))return {$number:String(x)};
  if(Object.is(x,-0))return {$number:'-0'};
  if(Buffer.isBuffer(x)||ArrayBuffer.isView(x))return {bytes:x.byteLength,sha256:sha(Buffer.from(x.buffer,x.byteOffset,x.byteLength))};
  if(Array.isArray(x))return x.map(normal);
  if(x&&typeof x==='object')return Object.fromEntries(Object.keys(x).map(k=>[k,normal(x[k])]));
  return x;
}
