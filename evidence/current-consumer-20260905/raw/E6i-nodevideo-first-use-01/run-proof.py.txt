def rows(paths):

        f=R/name
        assert f.is_file() and not f.is_symlink(),name

        b=f.read_bytes();result.append({'path':name,'bytes':len(b),'sha256':sha(b)})
    return result

save(runlabel+'-before.json',{'source':source,'dist':dist,'guards':guards,'harnessSha256':sha(script.read_bytes()),'approved18OwnerReceipt':{'path':str(approved_path),'sha256':sha(approved_path.read_bytes()),'sourceDigest':approved['sourceDigest']}})
with socket.socket()as check:
    assert check.connect_ex(('127.0.0.1',4983))!=0,'Port 4983 occupied; no process touched'
env=os.environ.copy()
removed=[]
for key in list(env):
    if key.startswith(('NODEVIDEO_','OPENAI_','OPENROUTER_','ANTHROPIC_','GEMINI_','HIGGSFIELD_')) or key=='CI':removed.append(key);env.pop(key)
env['NODEVIDEO_URL']='http://127.0.0.1:4983'
start=datetime.datetime.now(datetime.timezone.utc).isoformat()
owned=[]
receipt={'startedAt':start,'section':section,'cwd':str(R),'sourceCount':len(source),'distCount':len(dist),'childRemovedNamesOnly':removed,'childOverride':{'NODEVIDEO_URL':env['NODEVIDEO_URL']},'status':'RUNNING'}

        if section=='matrix':
