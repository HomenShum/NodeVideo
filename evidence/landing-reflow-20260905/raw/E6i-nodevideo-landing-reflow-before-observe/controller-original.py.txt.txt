import hashlib, json, os, subprocess, sys, time
from pathlib import Path
from datetime import datetime, timezone
import psutil

P = Path(__file__).resolve().parent
R = Path('D:/VSCode Projects/cafecorner_nodebench/nodebench_ai4/.portfolio-worktrees/nodevideo-current-consumer-20260905')
phase = sys.argv[1]
assert phase in ('before-build', 'before-observe', 'after-check', 'after-observe')
O = P / ('E6i-nodevideo-landing-reflow-' + phase)
O.mkdir()
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
def git(*args):
    return subprocess.check_output(['git', '--no-optional-locks', *args], cwd=R)
paths = [x.decode() for x in git('ls-files', '-z').split(b'\0') if x]
bindings = [{'path':x,'bytes':(R/x).stat().st_size,'sha256':sha(R/x)} for x in paths]
(O/'source-bindings.json').write_text(json.dumps(bindings, indent=2)+'\n', encoding='utf-8')
guard = {name:git(*args).decode() for name,args in {'head':('rev-parse','HEAD'),'branch':('branch','--show-current'),'status':('status','--porcelain=v1','--untracked-files=all'),'index':('ls-files','--stage')}.items()}
(O/'git-before.json').write_text(json.dumps(guard,indent=2)+'\n',encoding='utf-8')
commands = {'before-build':'npm run build', 'after-check':'npm run check', 'before-observe':'node "'+str(P/'E6i-observe-landing-reflow.mjs')+'" before', 'after-observe':'node "'+str(P/'E6i-observe-landing-reflow.mjs')+'" after'}
if phase in ('before-observe','after-observe'):
    for conn in psutil.net_connections(kind='tcp'):
        assert not (conn.status == psutil.CONN_LISTEN and conn.laddr.port == 4995), 'Port owned; do not interrupt'
env = {k:v for k,v in os.environ.items() if not k.upper().startswith(('OPENAI','ANTHROPIC','CONVEX','VITE_','NODEVIDEO','ELEVENLABS','GEMINI','GOOGLE_API','AZURE_OPENAI','VERCEL_')) and k.upper() not in ('NODE_OPTIONS','NODE_EXTRA_CA_CERTS','NODE_TLS_REJECT_UNAUTHORIZED','NODE_PRELOAD','BASH_ENV')}
start = time.monotonic()
known = {}
timed_out = False
with (O/'command.log').open('wb') as log:
    proc = subprocess.Popen(['cmd.exe','/d','/c',commands[phase]],cwd=R,env=env,stdout=log,stderr=subprocess.STDOUT,creationflags=subprocess.CREATE_NO_WINDOW)
    parent = psutil.Process(proc.pid)
    known[parent.pid] = parent.create_time()
    while proc.poll() is None:
        try:
            for child in parent.children(recursive=True):
                if len(known) < 512: known[child.pid] = child.create_time()
        except psutil.NoSuchProcess: pass
        if time.monotonic()-start > 300:
            timed_out = True
            break
        time.sleep(.5)
    if timed_out:
        for pid, created in reversed(list(known.items())):
            try:
                child = psutil.Process(pid)
                if child.create_time() == created: child.kill()
            except psutil.NoSuchProcess: pass
    code = proc.wait(timeout=10)
unchanged = all((R/x['path']).stat().st_size == x['bytes'] and sha(R/x['path']) == x['sha256'] for x in bindings)
after = {name:git(*args).decode() for name,args in {'head':('rev-parse','HEAD'),'branch':('branch','--show-current'),'status':('status','--porcelain=v1','--untracked-files=all'),'index':('ls-files','--stage')}.items()}
dist = [{'path':x.relative_to(R).as_posix(),'bytes':x.stat().st_size,'sha256':sha(x)} for x in sorted((R/'dist').rglob('*')) if x.is_file()]
(O/'dist-bindings.json').write_text(json.dumps(dist,indent=2)+'\n',encoding='utf-8')
record = {'status':'PASS' if code == 0 and not timed_out and unchanged and guard == after else 'FAIL', 'at':datetime.now(timezone.utc).isoformat(),'phase':phase,'command':commands[phase],'exitCode':code,'seconds':round(time.monotonic()-start,3),'budgetSeconds':300,'timedOut':timed_out,'sourceFiles':len(bindings),'sourceUnchanged':unchanged,'gitGuardUnchanged':guard==after,'distFiles':len(dist),'logSha256':sha(O/'command.log')}
(O/'receipt.json').write_text(json.dumps(record,indent=2)+'\n',encoding='utf-8')
print(json.dumps(record),flush=True)
assert record['status'] == 'PASS'
